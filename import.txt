// ---------------------------------------------------------------------
// STEP 0 (optional) - start from an empty database
// ---------------------------------------------------------------------
MATCH (n) DETACH DELETE n;


// ---------------------------------------------------------------------
// STEP 1 - Uniqueness constraints
// Each constraint replaces a relational PRIMARY KEY and automatically
// creates a backing RANGE index on the key property.
// ---------------------------------------------------------------------
CREATE CONSTRAINT customer_id IF NOT EXISTS
FOR (c:Customer) REQUIRE c.customerId IS UNIQUE;

CREATE CONSTRAINT product_id IF NOT EXISTS
FOR (p:Product) REQUIRE p.productId IS UNIQUE;

CREATE CONSTRAINT supplier_id IF NOT EXISTS
FOR (s:Supplier) REQUIRE s.supplierId IS UNIQUE;

CREATE CONSTRAINT location_id IF NOT EXISTS
FOR (l:Location) REQUIRE l.locationId IS UNIQUE;

CREATE CONSTRAINT category_name IF NOT EXISTS
FOR (cat:Category) REQUIRE cat.name IS UNIQUE;

CREATE CONSTRAINT sales_order_id IF NOT EXISTS
FOR (so:SalesOrder) REQUIRE so.salesOrderId IS UNIQUE;

CREATE CONSTRAINT purchase_order_id IF NOT EXISTS
FOR (po:PurchaseOrder) REQUIRE po.purchaseOrderId IS UNIQUE;

CREATE CONSTRAINT invoice_id IF NOT EXISTS
FOR (i:Invoice) REQUIRE i.invoiceId IS UNIQUE;

CREATE CONSTRAINT payment_id IF NOT EXISTS
FOR (pay:Payment) REQUIRE pay.paymentId IS UNIQUE;

CREATE CONSTRAINT delivery_id IF NOT EXISTS
FOR (d:Delivery) REQUIRE d.deliveryId IS UNIQUE;


// ---------------------------------------------------------------------
// STEP 2 - Additional indexes (secondary look-ups used by the queries)
// ---------------------------------------------------------------------
CREATE INDEX sales_order_status IF NOT EXISTS
FOR (so:SalesOrder) ON (so.status);

CREATE INDEX sales_order_date IF NOT EXISTS
FOR (so:SalesOrder) ON (so.orderDate);

CREATE INDEX purchase_order_status IF NOT EXISTS
FOR (po:PurchaseOrder) ON (po.status);

CREATE INDEX invoice_status IF NOT EXISTS
FOR (i:Invoice) ON (i.status);

CREATE INDEX delivery_status IF NOT EXISTS
FOR (d:Delivery) ON (d.status);

CREATE INDEX customer_name IF NOT EXISTS
FOR (c:Customer) ON (c.lastName, c.firstName);

CREATE INDEX location_type IF NOT EXISTS
FOR (l:Location) ON (l.locationType);

CREATE TEXT INDEX product_name_text IF NOT EXISTS
FOR (p:Product) ON (p.name);

// Relationship property index - supports low-stock look-ups (Query 1)
CREATE INDEX stock_quantity IF NOT EXISTS
FOR ()-[s:STOCKED_AT]-() ON (s.quantityOnHand);

// Wait until all indexes are online, then list them
CALL db.awaitIndexes(300);
SHOW INDEXES;


// ---------------------------------------------------------------------
// STEP 3 - Nodes (one CSV file per former relational table)
// IDs, phone numbers and postcodes are kept as strings/integers exactly
// as in SQL; dates are converted to native DATE values.
// ---------------------------------------------------------------------

// 3.1 Customer  (customers.csv -> CUSTOMER)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/customers.csv' AS row
MERGE (c:Customer {customerId: toInteger(row.CustomerID)})
SET c.firstName = row.FirstName,
    c.lastName  = row.LastName,
    c.name      = row.FirstName + ' ' + row.LastName,
    c.email     = row.Email,
    c.phone     = row.Phone,
    c.address   = row.Address,
    c.city      = row.City,
    c.postcode  = row.Postcode;

// 3.2 Category  (derived from PRODUCT.Category - see report section 2.2)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/products.csv' AS row
WITH DISTINCT row.Category AS categoryName
WHERE categoryName IS NOT NULL
MERGE (:Category {name: categoryName});

// 3.3 Product  (products.csv -> PRODUCT)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/products.csv' AS row
MERGE (p:Product {productId: toInteger(row.ProductID)})
SET p.name         = row.ProductName,
    p.description  = row.Description,
    p.unitPrice    = toFloat(row.UnitPrice),
    p.reorderLevel = toInteger(row.ReorderLevel);

// 3.4 Supplier  (suppliers.csv -> SUPPLIER)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/suppliers.csv' AS row
MERGE (s:Supplier {supplierId: toInteger(row.SupplierID)})
SET s.name        = row.SupplierName,
    s.contactName = row.ContactName,
    s.email       = row.Email,
    s.phone       = row.Phone,
    s.address     = row.Address,
    s.city        = row.City,
    s.postcode    = row.Postcode;

// 3.5 Location  (locations.csv -> LOCATION)
// The EER specialisation (Store / Warehouse) becomes an extra node label.
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/locations.csv' AS row
MERGE (l:Location {locationId: toInteger(row.LocationID)})
SET l.name         = row.LocationName,
    l.locationType = row.LocationType,
    l.address      = row.Address,
    l.city         = row.City,
    l.postcode     = row.Postcode
FOREACH (_ IN CASE WHEN row.LocationType = 'Store'     THEN [1] ELSE [] END | SET l:Store)
FOREACH (_ IN CASE WHEN row.LocationType = 'Warehouse' THEN [1] ELSE [] END | SET l:Warehouse);

// 3.6 SalesOrder  (sales_orders.csv -> SALES_ORDER)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/sales_orders.csv' AS row
MERGE (so:SalesOrder {salesOrderId: toInteger(row.SalesOrderID)})
SET so.orderDate   = date(row.OrderDate),
    so.status      = row.Status,
    so.totalAmount = toFloat(row.TotalAmount);

// 3.7 PurchaseOrder  (purchase_orders.csv -> PURCHASE_ORDER)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/purchase_orders.csv' AS row
MERGE (po:PurchaseOrder {purchaseOrderId: toInteger(row.PurchaseOrderID)})
SET po.orderDate    = date(row.OrderDate),
    po.expectedDate = CASE WHEN row.ExpectedDate IS NULL OR row.ExpectedDate = ''
                           THEN null ELSE date(row.ExpectedDate) END,
    po.status       = row.Status,
    po.totalAmount  = toFloat(row.TotalAmount);

// 3.8 Invoice  (invoices.csv -> INVOICE)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/invoices.csv' AS row
MERGE (i:Invoice {invoiceId: toInteger(row.InvoiceID)})
SET i.invoiceDate = date(row.InvoiceDate),
    i.totalAmount = toFloat(row.TotalAmount),
    i.status      = row.Status;

// 3.9 Payment  (payments.csv -> PAYMENT)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/payments.csv' AS row
MERGE (pay:Payment {paymentId: toInteger(row.PaymentID)})
SET pay.paymentDate   = date(row.PaymentDate),
    pay.amount        = toFloat(row.Amount),
    pay.paymentMethod = row.PaymentMethod,
    pay.status        = row.Status;

// 3.10 Delivery  (deliveries.csv -> DELIVERY)
// DeliveryDate is NULL for pending deliveries, so it is only set if present.
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/deliveries.csv' AS row
MERGE (d:Delivery {deliveryId: toInteger(row.DeliveryID)})
SET d.deliveryDate   = CASE WHEN row.DeliveryDate IS NULL OR row.DeliveryDate = ''
                            THEN null ELSE date(row.DeliveryDate) END,
    d.carrier        = row.Carrier,
    d.trackingNumber = row.TrackingNumber,
    d.status         = row.Status;


// ---------------------------------------------------------------------
// STEP 4 - Relationships
// Foreign keys become relationships; the three associative tables
// (SALES_ORDER_ITEM, PURCHASE_ORDER_ITEM, STOCK) become relationships
// that carry the former non-key columns as properties.
// ---------------------------------------------------------------------

// 4.1 (:Product)-[:IN_CATEGORY]->(:Category)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/products.csv' AS row
MATCH (p:Product {productId: toInteger(row.ProductID)})
MATCH (cat:Category {name: row.Category})
MERGE (p)-[:IN_CATEGORY]->(cat);

// 4.2 (:Customer)-[:PLACED]->(:SalesOrder)      [SALES_ORDER.CustomerID FK]
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/sales_orders.csv' AS row
MATCH (c:Customer {customerId: toInteger(row.CustomerID)})
MATCH (so:SalesOrder {salesOrderId: toInteger(row.SalesOrderID)})
MERGE (c)-[:PLACED]->(so);

// 4.3 (:SalesOrder)-[:CONTAINS]->(:Product)       [SALES_ORDER_ITEM]
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/sales_order_items.csv' AS row
MATCH (so:SalesOrder {salesOrderId: toInteger(row.SalesOrderID)})
MATCH (p:Product {productId: toInteger(row.ProductID)})
MERGE (so)-[ci:CONTAINS]->(p)
SET ci.quantity  = toInteger(row.Quantity),
    ci.unitPrice = toFloat(row.UnitPrice),
    ci.lineTotal = round(toInteger(row.Quantity) * toFloat(row.UnitPrice), 2);

// 4.4 (:PurchaseOrder)-[:PLACED_WITH]->(:Supplier) [PURCHASE_ORDER.SupplierID FK]
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/purchase_orders.csv' AS row
MATCH (po:PurchaseOrder {purchaseOrderId: toInteger(row.PurchaseOrderID)})
MATCH (s:Supplier {supplierId: toInteger(row.SupplierID)})
MERGE (po)-[:PLACED_WITH]->(s);

// 4.5 (:PurchaseOrder)-[:ORDERS]->(:Product)      [PURCHASE_ORDER_ITEM]
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/purchase_order_items.csv' AS row
MATCH (po:PurchaseOrder {purchaseOrderId: toInteger(row.PurchaseOrderID)})
MATCH (p:Product {productId: toInteger(row.ProductID)})
MERGE (po)-[oi:ORDERS]->(p)
SET oi.quantity  = toInteger(row.Quantity),
    oi.unitCost  = toFloat(row.UnitCost),
    oi.lineTotal = round(toInteger(row.Quantity) * toFloat(row.UnitCost), 2);

// 4.6 (:Product)-[:STOCKED_AT]->(:Location)       [STOCK]
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/stock.csv' AS row
MATCH (p:Product {productId: toInteger(row.ProductID)})
MATCH (l:Location {locationId: toInteger(row.LocationID)})
MERGE (p)-[st:STOCKED_AT]->(l)
SET st.quantityOnHand = toInteger(row.QuantityOnHand),
    st.lastUpdated    = date(row.LastUpdated);

// 4.7 (:SalesOrder)-[:INVOICED_AS]->(:Invoice)    [INVOICE.SalesOrderID FK, 1:0..1]
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/invoices.csv' AS row
MATCH (so:SalesOrder {salesOrderId: toInteger(row.SalesOrderID)})
MATCH (i:Invoice {invoiceId: toInteger(row.InvoiceID)})
MERGE (so)-[:INVOICED_AS]->(i);

// 4.8 (:Payment)-[:PAYS]->(:Invoice)              [PAYMENT.InvoiceID FK]
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/payments.csv' AS row
MATCH (pay:Payment {paymentId: toInteger(row.PaymentID)})
MATCH (i:Invoice {invoiceId: toInteger(row.InvoiceID)})
MERGE (pay)-[:PAYS]->(i);

// 4.9 (:SalesOrder)-[:SHIPPED_BY]->(:Delivery)    [DELIVERY.SalesOrderID FK]
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/xhhetri/-PRT563-ADVANCED-DATA-MANAGEMENT/refs/heads/main/csv/deliveries.csv' AS row
MATCH (so:SalesOrder {salesOrderId: toInteger(row.SalesOrderID)})
MATCH (d:Delivery {deliveryId: toInteger(row.DeliveryID)})
MERGE (so)-[:SHIPPED_BY]->(d);


// ---------------------------------------------------------------------
// STEP 5 - Derived (summary) relationships used by analytics
// These shortcut the 2-hop order paths so that GDS projections and
// recommendation queries can work on direct links.
// ---------------------------------------------------------------------

// 5.1 (:Customer)-[:BOUGHT]->(:Product)
MATCH (c:Customer)-[:PLACED]->(so:SalesOrder)-[ci:CONTAINS]->(p:Product)
WITH c, p, sum(ci.quantity) AS units, count(DISTINCT so) AS orders
MERGE (c)-[b:BOUGHT]->(p)
SET b.totalQuantity = units,
    b.orderCount    = orders;

// 5.2 (:Supplier)-[:SUPPLIES]->(:Product)
MATCH (s:Supplier)<-[:PLACED_WITH]-(po:PurchaseOrder)-[oi:ORDERS]->(p:Product)
WITH s, p, sum(oi.quantity) AS units, count(DISTINCT po) AS orders,
     sum(oi.quantity * oi.unitCost) AS spend
MERGE (s)-[r:SUPPLIES]->(p)
SET r.totalQuantity = units,
    r.orderCount    = orders,
    r.avgUnitCost   = round(spend / units, 2);


// ---------------------------------------------------------------------
// STEP 6 - Verification
// ---------------------------------------------------------------------
// Node counts per label (expected: Customer 5, Product 8, Supplier 4,
// Location 4, Category 5, SalesOrder 6, PurchaseOrder 5, Invoice 6,
// Payment 4, Delivery 6)
MATCH (n)
RETURN labels(n) AS labels, count(*) AS nodes
ORDER BY nodes DESC;

// Relationship counts per type (expected: STOCKED_AT 24, ORDERS 12,
// CONTAINS 10, BOUGHT 10, SUPPLIES 10, IN_CATEGORY 8, PLACED 6,
// INVOICED_AS 6, SHIPPED_BY 6, PLACED_WITH 5, PAYS 4)
MATCH ()-[r]->()
RETURN type(r) AS relationship, count(*) AS total
ORDER BY total DESC;

// Visualise the schema (meta-graph) of the imported database
CALL db.schema.visualization();
